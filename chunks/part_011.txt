    grad.addColorStop(0.5,`hsla(${p.hue},70%,55%,${0.15*p.depth})`);
      grad.addColorStop(1,'transparent');
    } else {
      grad.addColorStop(0,`hsla(${p.hue},70%,60%,${0.25*p.depth})`);
      grad.addColorStop(1,'transparent');
    }
    sCtx.beginPath();sCtx.arc(p.x,p.y,glow*1.8,0,Math.PI*2);
    sCtx.save();
    sCtx.shadowBlur=p.speaking?28*p.depth:14*p.depth;
    sCtx.shadowColor=p.ghost?`rgba(180,79,255,${0.5*p.depth})`:`hsla(${p.hue},80%,65%,${0.6*p.depth})`;
    sCtx.fillStyle=grad;sCtx.fill();
    sCtx.restore();

    // Cercle
    sCtx.beginPath();sCtx.arc(p.x,p.y,r,0,Math.PI*2);
    sCtx.strokeStyle=p.ghost?`rgba(180,79,255,${0.5*p.depth})`:`hsla(${p.hue},75%,65%,${0.7*p.depth})`;
    sCtx.lineWidth=p.speaking?2:1.2;sCtx.stroke();

    // Fond cercle
    const fill=sCtx.createRadialGradient(p.x,p.y-r*0.2,0,p.x,p.y,r);
    fill.addColorStop(0,p.ghost?`rgba(180,79,255,${0.15*p.depth})`:`hsla(${p.hue},50%,20%,${0.3*p.depth})`);
    fill.addColorStop(1,`rgba(0,0,0,${0.3*p.depth})`);
    sCtx.beginPath();sCtx.arc(p.x,p.y,r,0,Math.PI*2);
    sCtx.fillStyle=fill;sCtx.fill();

    // Emoji
    sCtx.font=`${r*0.9}px Arial`;sCtx.textAlign='center';sCtx.textBaseline='middle';
    sCtx.globalAlpha=p.ghost?0.3*p.depth:0.85*p.depth;
    sCtx.fillText(p.humeur,p.x,p.y);sCtx.globalAlpha=1;

    // Nom + ville
    if(!p.ghost&&p.depth>0.7){
      sCtx.font=`${10*p.depth}px Share Tech Mono,monospace`;
      sCtx.fillStyle=`rgba(255,255,255,${0.55*p.depth})`;
      sCtx.textAlign='center';sCtx.textBaseline='top';
      sCtx.fillText(`${p.prenom}·${p.ville}`,p.x,p.y+r+4*p.depth);
    }

    // Onde message
    if(p.speaking){
      for(let w=1;w<=3;w++){
        const wr=r+w*12+Math.sin(p.pulse*2+w)*3;
        const wop=(1-w/4)*0.3*p.depth;
        sCtx.beginPath();sCtx.arc(p.x,p.y,wr,0,Math.PI*2);
        sCtx.strokeStyle=`hsla(${p.hue},80%,70%,${wop})`;
        sCtx.lineWidth=0.8;sCtx.stroke();
      }
    }
  });

  requestAnimationFrame(drawSalon);
}

function toggleMusique(){
  const s=MUSIC_STYLES[musicStyle];
  if(!s.src){showError('⚠ Fichier audio pas encore ajouté pour ce style.');return;}
  if(bgAudio.paused){
    if(!bgAudio.src)bgAudio.src=s.src;
    bgAudio.play().then(majMusicUI).catch(()=>{
      showError('⚠ Le navigateur bloque la lecture auto. Retape sur ▶.');
    });
  }else{
    bgAudio.pause();
    majMusicUI();
  }
}
