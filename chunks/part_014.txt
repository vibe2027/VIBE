async function adminLoadStats(){
  try{
    const [ins,sig,msg,sos]=await Promise.all([
      _supa.from('landing_inscriptions').select('id',{count:'exact',head:true}),
      _supa.from('tribunal_signalements').select('id',{count:'exact',head:true}),
      _supa.from('salon_messages').select('id',{count:'exact',head:true}),
      _supa.from('ange_alerts').select('id',{count:'exact',head:true}),
    ]);
    const el=document.getElementById('admin-stats');
    if(el)el.innerHTML=`Inscrits: <b style="color:#D4AF37">${ins.count||0}</b> / ${MAX_FOUNDERS} billets<br>Places restantes: <b style="color:#D4AF37">${MAX_FOUNDERS-(ins.count||0)}</b><br>Signalements: ${sig.count||0} · Messages: ${msg.count||0} · SOS: ${sos.count||0}`;
  }catch(e){}
}
