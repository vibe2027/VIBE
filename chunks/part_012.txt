async function sendChatMsg(){

  const msg=inp.value.trim();if(!msg)return;

  if(monProfil?.lecture_seule_jusqu && new Date(monProfil.lecture_seule_jusqu) > new Date()){
    const dateFin=new Date(monProfil.lecture_seule_jusqu).toLocaleDateString('fr-CA');

    return;
  }
  inp.value='';
  const estFantome=currentSalon==='fantomes';
  try{
    await _supa.from('salon_messages').insert({
      prenom:estFantome?'👻':(CURRENT_USER.user_metadata?.prenom||'Membre'),
      ville:estFantome?'':(CURRENT_USER.user_metadata?.ville||'QC'),

    });
    // L'utilisateur parle — animer son profil
    if(!estFantome&&salProfs.length){salProfs[0].speaking=true;salProfs[0].speakTimer=150;}
  }catch(e){showError('⚠ Erreur envoi.');}
}

function openLogin(){document.getElementById('login-modal').classList.add('open')}
function closeLogin(){document.getElementById('login-modal').classList.remove('open')}
document.getElementById('login-modal').addEventListener('click',e=>{if(e.target===e.currentTarget)closeLogin()});
document.addEventListener('keydown',e=>{if(e.key==='Escape')closeLogin()});

async function handleLogin(e){
  e.preventDefault();
  const email=document.getElementById('login-email').value.trim();
  const mdp=document.getElementById('login-mdp').value;
  const btn=document.querySelector('#login-form .submit-btn');
  if(btn){btn.disabled=true;btn.querySelector('span').textContent='Connexion...';}
  try{
    const {data,error}=await _supa.auth.signInWithPassword({email,password:mdp});
    if(error)throw error;
    closeLogin();
    showSuccess(`✦ Bienvenue ${data.user?.user_metadata?.prenom||'Membre'} !`);
    setTimeout(()=>location.reload(),1000);
  }catch(err){
    if(btn){btn.disabled=false;btn.querySelector('span').textContent='Entrer dans VIBE ✦';}
    showError('⚠ '+(err.message?.includes('Invalid')?'Courriel ou mot de passe incorrect.':err.message||'Erreur.'));
  }
}

async function handleForgotPassword(){
  const email=document.getElementById('login-email').value.trim();
  if(!email){showError('⚠ Entre ton courriel d\'abord.');return;}
  try{
    await _supa.auth.resetPasswordForEmail(email,{redirectTo:'https://vibegay.ca'});
    showSuccess('✦ Courriel de réinitialisation envoyé !');closeLogin();
  }catch(e){showError('⚠ '+(e.message||'Erreur.'));}
}

async function handleMagicLink(){
  const email=document.getElementById('login-email').value.trim();
  if(!email){showError('⚠ Entre ton courriel d\'abord.');return;}
  try{
    const {error}=await _supa.auth.signInWithOtp({email,options:{emailRedirectTo:'https://vibegay.ca'}});
    if(error)throw error;
    showSuccess('✦ Lien envoyé à '+email+' — clique dessus pour entrer directement.');closeLogin();
  }catch(e){showError('⚠ '+(e.message||'Erreur.'));}
}
