tut:'active',
      notes:`GPS:${gpsData.lat},${gpsData.lng}|Acc:${gpsData.accuracy||'N/A'}`
    });
    if(error)throw error;
  }catch(e){
    angelSaveFailed=true;
    console.error('Mode Ange: echec envoi alerte',e);
  }

  // Flash rouge sur l'écran
  const flash=document.createElement('div');
  flash.style.cssText='position:fixed;inset:0;z-index:5000;background:rgba(255,0,0,0.08);pointer-events:none;animation:sos-flash 0.6s ease-in-out 3';
  const style=document.createElement('style');
  style.textContent='@keyframes sos-flash{0%,100%{opacity:0}50%{opacity:1}}';
  document.head.appendChild(style);
  document.body.appendChild(flash);
  setTimeout(()=>{flash.remove();style.remove()},2000);

  if(angelSaveFailed){
    txt.textContent='⚠ Echec de l\'envoi — reessaie';
    setAngelStatus('flow','ECHEC envoi alerte','alert');
    showError('⚠ Mode Ange : l\'alerte n\'a PAS pu etre envoyee. Reessaie ou contacte les secours directement.');
  }else{
    txt.textContent='✦ Ange gardien activé — aide en route';
    setAngelStatus('flow','Alerte envoyée','alert');
    showSuccess('✦ Mode Ange activé — alerte envoyée !');
  }
  loadStats();
  setTimeout(()=>{
    txt.textContent='⚡ Activer Mode Ange Gardien';
    btn.style.borderColor='';btn.style.color='';sosActive=false;
    setAngelStatus('flow','Prêt','ready');
    if(gpsData.accuracy)setAngelStatus('gps','GPS OK · '+gpsData.accuracy,'ready');
  },8000);
}

async function handleSignalement(e){
  e.preventDefault();
  if(IS_ADMIN){showError('⚠ Compte admin — immunisé du tribunal.');return;}
  const pseudo=document.getElementById('trib-pseudo').value.trim()||'Anonyme';
  const accuse=document.getElementById('trib-accuse').value.trim();
  const motif=document.getElementById('trib-motif').value;
  const details=document.getElementById('trib-details').value.trim();
  const btn=e.target.querySelector('.trib-submit');
  const suc=document.getElementById('trib-success');
  if(!motif){showError('⚠ Choisis un motif.');return;}
  if(!accuse){showError('⚠ Indique le pseudo du membre visé.');return;}
  if(btn){btn.disabled=true;btn.textContent='Envoi...';}
  try{
    await _supa.from('tribunal_signalements').insert({pseudo_reporter:pseudo.slice(0,30),accused_username:accuse.slice(0,30),motif,details:details.slice(0,500)||null,statut:'en_examen'});
    document.getElementById('tribunal-form').style.display='none';
    suc.style.display='block';
    suc.innerHTML='✦ Signalement soumis — le jury examine dans les 24h.';
    loadStats();
  }catch(err){
    if(btn){btn.disabled=false;btn.textContent='Soumettre au tribunal ↗';}
    showError('⚠ '+(err.message||'Erreur.'));
  }
}
