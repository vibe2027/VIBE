      box.addEventListener('pointerdown',devoiler);
      box.addEventListener('pointerup',refloutter);

  }catch(e){showError('⚠ Photo pas envoyée.');}
}
async function envoyerPrive(){
  const inp=document.getElementById('prive-input');
  const txt=inp.value.trim();if(!txt||!priveAvec)return;
  const {data:monProfilRO}=await _supa.from('profiles').select('lecture_seule_jusqu').eq('id',CURRENT_USER.id).maybeSingle();
  if(monProfilRO?.lecture_seule_jusqu && new Date(monProfilRO.lecture_seule_jusqu) > new Date()){

    return;
  }
  inp.value='';
  try{
    let contenu_traduit=null,langue_cible=null;
    try{
      const {data:langueDest}=await _supa.rpc('obtenir_langue_membre',{p_id:priveAvec});
      const {data:monProfil}=await _supa.from('profiles').select('langue').eq('id',CURRENT_USER.id).maybeSingle();
      const mesLangue=monProfil?.langue||'fr';
      langue_cible=langueDest||'fr';
      if(langue_cible&&langue_cible!==mesLangue){
        const {data:tr}=await _supa.functions.invoke('translate-message',{body:{texte:txt,langue_cible}});
        if(tr?.traduit)contenu_traduit=tr.traduit;
      }
    }catch(e){/* traduction indisponible — le message original part quand même */}
    const {error}=await _supa.from('messages_prives').insert({
      expediteur:CURRENT_USER.id,destinataire:priveAvec,
      expediteur_pseudo:CURRENT_USER.user_metadata?.prenom||'Membre',
      contenu:txt,contenu_traduit,langue_cible
    });
    if(error)throw error;
  }catch(e){showError('⚠ Message pas envoyé.');}
}
