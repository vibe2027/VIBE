async function adminOffrirAcces(btn){
  const email=document.getElementById('acces-email').value.trim();
  const mois=parseInt(document.getElementById('acces-mois').value,10)||0;
  if(!email){showError('⚠ Courriel requis.');return;}
  btn.disabled=true;btn.textContent='...';
  try{
    const {data:rows,error:eSearch}=await _supa.from('profiles').select('id,real_email').eq('real_email',email).limit(1);
    if(eSearch||!rows?.length){showError('⚠ Membre introuvable.');btn.disabled=false;btn.textContent='🎁 OFFRIR';return;}
    const {data,error}=await _supa.rpc('admin_offrir_acces',{p_id:rows[0].id,p_mois:mois});
    if(error||!data?.ok){showError('⚠ '+(data?.error||'Erreur.'));btn.disabled=false;btn.textContent='🎁 OFFRIR';return;}
    showSuccess('✦ Accès offert !');btn.textContent='✓ FAIT';
  }catch(e){showError('⚠ Erreur.');btn.disabled=false;btn.textContent='🎁 OFFRIR';}
}
