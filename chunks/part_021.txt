function previsPhoto(inp){

  const r=new FileReader();

async function sauverProfil(e){
  e.preventDefault();
  const btn=document.getElementById('pf-save');btn.disabled=true;btn.textContent='Enregistrement...';
  try{
    let photo_url=null;
    const f=document.getElementById('pf-photo').files[0];
    if(f){
      if(f.size>5*1024*1024)throw new Error('Photo trop lourde (max 5 Mo).');
      const path=CURRENT_USER.id+'/avatar_'+Date.now()+'.'+(f.name.split('.').pop()||'jpg');
      const {error:upErr}=await _supa.storage.from('avatars').upload(path,f,{upsert:true});
      if(upErr)throw upErr;
      photo_url=_supa.storage.from('avatars').getPublicUrl(path).data.publicUrl;
    }
    const age=parseInt(document.getElementById('pf-age').value,10);
    if(!age||age<18)throw new Error('18 ans et plus.');
    const row={
      id:CURRENT_USER.id,
      display_name:document.getElementById('pf-nom').value.trim().slice(0,30),
      age:age,
      city:document.getElementById('pf-ville').value.trim().slice(0,40),
      bio:document.getElementById('pf-bio').value.trim().slice(0,300),
      langue:document.getElementById('pf-langue').value,
      partage_distance:document.getElementById('pf-partage-distance').checked,
      intention:pfIntention,
      identites:pfIdentites,
      is_online:true,
      updated_at:new Date().toISOString()
    };
    if(photo_url)row.photo_url=photo_url;
    const {error}=await _supa.from('profiles').upsert(row);
    if(error)throw error;
    document.getElementById('profil-modal').classList.remove('open');
    showSuccess('✦ Profil enregistré !');
    chargerDecouverte();
  }catch(err){showError('⚠ '+(err.message||'Erreur.'));}
  btn.disabled=false;btn.textContent='Enregistrer mon profil ✦';
}

let maPosition=null,maPositionTentee=false;
async function obtenirMaPosition(){
  if(maPosition||maPositionTentee)return maPosition;
  maPositionTentee=true;
  try{
    const pos=await new Promise((res,rej)=>navigator.geolocation.getCurrentPosition(res,rej,{timeout:6000,maximumAge:600000}));
    maPosition={lat:pos.coords.latitude,lon:pos.coords.longitude};
    if(CURRENT_USER)_supa.from('profiles').update({latitude:maPosition.lat,longitude:maPosition.lon}).eq('id',CURRENT_USER.id).then(()=>{});
  }catch(e){maPosition=null;}
  return maPosition;
}
